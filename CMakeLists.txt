cmake_minimum_required(VERSION 3.16)
project(FixtureGraphicsLab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FIXTURE_BUILD_VIEWER "Build the OpenGL 3.3 viewer" ON)
add_library(fixture_core src/geometry.cpp)
target_include_directories(fixture_core PUBLIC include)
if(MSVC)
  target_compile_options(fixture_core PRIVATE /W4)
else()
  target_compile_options(fixture_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(fixture_cli src/cli.cpp)
target_link_libraries(fixture_cli PRIVATE fixture_core)
enable_testing()
add_executable(fixture_tests tests/test_geometry.cpp)
target_link_libraries(fixture_tests PRIVATE fixture_core)
add_test(NAME geometry_and_transforms COMMAND fixture_tests)
if(FIXTURE_BUILD_VIEWER)
  add_executable(fixture_viewer src/viewer.cpp)
  target_link_libraries(fixture_viewer PRIVATE fixture_core)
  if(WIN32)
    target_sources(fixture_viewer PRIVATE src/window_win32.cpp)
    target_link_libraries(fixture_viewer PRIVATE opengl32 gdi32 user32)
  else()
    find_package(OpenGL REQUIRED)
    find_package(glfw3 3.3 REQUIRED)
    target_sources(fixture_viewer PRIVATE src/window_glfw.cpp)
    target_link_libraries(fixture_viewer PRIVATE OpenGL::GL glfw)
  endif()
endif()
